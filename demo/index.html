import { IPod } from "ipod-simulator/react";
import {
  defaultSongs, defaultPlaylists,
  type Song, type Playlist,
} from "ipod-simulator";

const songs: Song[] = [...defaultSongs, {
  id: "my-song",
  name: "My song",
  artistName: "My artist",
  previewUrl: "/music/my-song.mp3",
}];

const playlists: Playlist[] = [...defaultPlaylists, {
  id: "favorites",
  name: "Favorites",
  songIds: ["my-song"],
}];

export default function App() {
  return <IPod songs={songs} playlists={playlists} />;
}
